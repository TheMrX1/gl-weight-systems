cmake_minimum_required(VERSION 3.24)
project(gl_weight_systems LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(BOOST_INCLUDE_DIR boost/json/src.hpp HINTS ${BOOST_ROOT} ENV BOOST_ROOT)
if(NOT BOOST_INCLUDE_DIR)
  message(FATAL_ERROR "Install Boost headers (including Boost.JSON), or pass -DBOOST_ROOT=/path/to/boost")
endif()
include_directories(cpp/include ${BOOST_INCLUDE_DIR})
add_library(gl_core cpp/src/chord_diagram.cpp cpp/src/gl_weight.cpp)
add_library(gl_symbolic cpp/src/symbolic_gl.cpp)
target_link_libraries(gl_symbolic PUBLIC gl_core)
foreach(app evaluate_word audit_permutations)
  add_executable(${app} cpp/apps/${app}.cpp)
  target_link_libraries(${app} PRIVATE gl_core)
endforeach()
foreach(app symbolic_witness gl_polynomial primitive_gl)
  add_executable(${app} cpp/apps/${app}.cpp)
  target_link_libraries(${app} PRIVATE gl_symbolic)
endforeach()
foreach(app representation_weights hook_difference audit_direct_gl4 mutation_tensor graph_corank graph_diagonal graph_pairs sl2_direct_matrix chromatic)
  add_executable(${app} cpp/apps/${app}.cpp)
endforeach()
foreach(app sl2_word_batch sl2_primitive)
  add_executable(${app} cpp/apps/${app}.cpp)
  target_link_libraries(${app} PRIVATE gl_core)
endforeach()
add_executable(rectangle_certificate cpp/apps/rectangle_certificate.cpp)
target_link_libraries(rectangle_certificate PRIVATE gl_core)
add_executable(symbolic_traceless_compressed EXCLUDE_FROM_ALL cpp/apps/symbolic_compressed.cpp cpp/src/symbolic_gl_compressed_rank.cpp)
target_compile_definitions(symbolic_traceless_compressed PRIVATE CHORD_SYMBOLIC_TRACELESS)
target_link_libraries(symbolic_traceless_compressed PRIVATE gl_core)
add_executable(symbolic_low_memory cpp/apps/symbolic_compressed.cpp cpp/src/symbolic_gl_low_memory.cpp)
target_link_libraries(symbolic_low_memory PRIVATE gl_core)
configure_file(results/hooks/universal_delta.json universal_delta.json COPYONLY)
add_library(share_core cpp/src/sl4_share_detector.cpp)
target_link_libraries(share_core PUBLIC gl_core)
foreach(app verify_4_7_witness tensor_pair_search)
 add_executable(${app} cpp/apps/${app}.cpp)
 target_link_libraries(${app} PRIVATE share_core)
endforeach()
find_package(Python3 COMPONENTS Interpreter)
enable_testing()
if(Python3_FOUND)
 add_test(NAME results_exact COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/verify_results.py --build-dir ${CMAKE_BINARY_DIR})
 add_test(NAME hook_identity COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/check_sparse_identity.py)
 add_test(NAME small_rank COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/check_small_rank.py)
endif()
